       a.status,
       p.hospital_no,
       p.first_name || ' ' || p.last_name AS patient
FROM hospital.appointments a
JOIN hospital.patients p ON p.patient_id = a.patient_id
WHERE a.doctor_id = 23
  AND a.scheduled_start >= '2026-11-04'
